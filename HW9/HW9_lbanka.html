Standardizing Weather Variables

I standardized the numeric weather predictors using the means and standard deviations from the training dataset.

weather_vars <- c(
  "temp",
  "humidity",
  "wind_speed",
  "visibility",
  "dew_point_temp",
  "solar_radiation",
  "rainfall",
  "snowfall"
)

train_means <- bike_train |>
  summarise(across(all_of(weather_vars), mean))

train_sds <- bike_train |>
  summarise(across(all_of(weather_vars), sd))

for (v in weather_vars) {
  bike_train[[paste0(v, "_z")]] <-
    (bike_train[[v]] - train_means[[v]]) / train_sds[[v]]

  bike_test[[paste0(v, "_z")]] <-
    (bike_test[[v]] - train_means[[v]]) / train_sds[[v]]
}

Comparing Model Performance

I calculated the RMSE for each model using the testing dataset. A lower RMSE indicates better predictive performance.

rmse <- function(actual, predicted) {
  sqrt(mean((actual - predicted)^2))
}

pred1 <- predict(model1, newdata = bike_test)
pred2 <- predict(model2, newdata = bike_test)
pred3 <- predict(model3, newdata = bike_test)

model_results <- tibble(
  model = c("Model 1", "Model 2", "Model 3"),
  test_rmse = c(
    rmse(bike_test$bike_count, pred1),
    rmse(bike_test$bike_count, pred2),
    rmse(bike_test$bike_count, pred3)
  )
)
