Let’s standardize/normalize numeric variables

num_cols <- c("temp","humidity","wind_speed","visibility","dew_point_temp","solar_radiation","rainfall","snowfall")

train_means <- sapply(bike_train[num_cols], mean)
train_sds <- sapply(bike_train[num_cols], sd)

standardize_with <- function(df, means, sds, cols) {
  for(col in cols) {
    df[[paste0(col, "_std")]] <- (df[[col]] - means[[col]]) / sds[[col]]
  }
  df
}

bike_train <- standardize_with(bike_train, train_means, train_sds, num_cols)
bike_test <- standardize_with(bike_test, train_means, train_sds, num_cols)

Let’s write a function that does test-set RMSE for reach

rmse <- function(model, newdata) {
  preds <- predict(model, newdata = newdata)
  sqrt(mean((newdata$rented_count - preds) ^ 2))
}
