EDA

Same cleanup as HW9: snake_case names, parse date, factor the three categorial columns, drop the non-functioning hours, then aggregate hourly to daily.

bike <- bike_data 
names(bike) <- c("date","rented_count","hour","temp", "humidity","wind_speed","visibility","dew_point_temp","solar_radiation",
                 "rainfall","snowfall","seasons","holiday","functioning_day")

bike <- bike |>
  mutate(date = dmy(date)) |>
  mutate(across(c(seasons, holiday, functioning_day), as.factor)) |>
  filter(functioning_day == "Yes")

daily <- bike |>
  group_by(date, seasons, holiday) |>
  summarise(
    rented_count = sum(rented_count),
    rainfall = sum(rainfall),
    snowfall = sum(snowfall),
    temp = mean(temp),
    humidity = mean(humidity),
    wind_speed = mean(wind_speed),
    visibility = mean(visibility),
    dew_point_temp = mean(dew_point_temp),
    solar_radiation = mean(solar_radiation),
    .groups = "drop"

Fitting MLR Models

Recipes

Recipe 1:

  • derive a weekday/weekend factor from date
  • step_date creates brand new column that is date_dow, original date column remains untouched
  • use date_dow to further refine the factor to weekend and weekday
  • all numeric variables are normalized
  • step_dummy to create k-1 columns per each of the categorical column where k is number of factor levels
MLR_recipe1 <- recipe(rented_count ~ ., data = bike_train) |>
  update_role(date, new_role = "id") |>
  step_date(date, features = "dow") |>
  step_mutate(day_type =  factor(if_else(date_dow %in% c("Sat","Sun"), "weekend", "weekday"))) |>
  step_rm(date_dow) |>
  step_normalize(all_numeric_predictors()) |>
  step_dummy(seasons, holiday, day_type)

Recipe 2:

  • all the steps of recipe 1
  • Add in interactions between seasons and holiday, seasons and temp, temp and rainfall using starts_with() because step_dummy crates k-1 columns for seasons that start with seasons
MLR_recipe2 <- recipe(rented_count ~ ., data = bike_train) |>

Recipe 3:

  • all the steps of recipe 2
  • Add in quadratic terms for the standardized temp and humidity variables.
MLR_recipe3 <- recipe(rented_count ~ ., data = bike_train) |>
  update_role(date, new_role = "id") |>
  step_date(date, features = "dow") |>
  step_mutate(day_type =  factor(if_else(date_dow %in% c("Sat","Sun"), "weekend", "weekday"))) |>
  step_rm(date_dow) |>

  step_dummy(seasons, holiday, day_type) |>
  step_interact(~starts_with("seasons"):starts_with("holiday") + 
                 starts_with("seasons"):temp +
                 temp:rainfall) |>
  step_mutate(tetmp_sq = temp^2, humidity_sq = humidity^2)
